% TikZ system architecture diagram — no background layer needed.
\tikzset{
    block/.style = {
        draw, fill=white, rectangle, minimum height=3em, minimum width=6em,
        rounded corners, align=center
    },
    arrow/.style = {->, >=stealth, thick},
    line/.style = {-, thick},
    highlight/.style = {draw=red, line width=1.5pt, fill=red!10}
}

\begin{tikzpicture}[node distance=1.5cm and 2cm, auto]
    % --- Environment Nodes (Left Side) ---
    \node[block, fill=gray!10] (network) {Network Channel\\(Throughput $C_k$)};
    \node[block, fill=gray!10, below=of network] (buffer) {Client Video Player\\(Buffer $B_k$)};
    \node[block, highlight, below=of buffer] (video) {\textbf{Video Source}\\(Next Chunk Sizes $\vec{v}_{next}$)};

    % Environment boundary (drawn behind, no pgfonlayer needed)
    \node[draw=blue!50!black, fill=blue!5, dashed, inner sep=15pt, rounded corners,
          fit=(network) (buffer) (video),
          label={[anchor=north, font=\bfseries, text=blue!50!black]north:ABR Streaming Environment}] (env_boundary) {};

    % --- Agent Nodes (Right Side) ---
    \node[block, right=6cm of buffer, fill=red!15] (actor) {Actor Network\\Policy $\pi_\theta(a|s)$};
    \node[block, below=of actor, fill=blue!15] (critic) {Critic Network\\Value $V_\phi(s)$};

    % Agent boundary
    \node[draw=red!50!black, fill=red!5, dashed, inner sep=15pt, rounded corners,
          fit=(actor) (critic),
          label={[anchor=north, font=\bfseries, text=red!50!black]north:Future-Aware PPO Agent}] (agent_boundary) {};

    % --- Connections (Arrows) ---
    \coordinate (midpoint_state) at ($(env_boundary.east)!0.5!(agent_boundary.west) + (0, 1.5)$);
    \draw[line, gray] (network.east) -- ++(1,0) |- (midpoint_state);
    \draw[line, gray] (buffer.east) -- ++(1,0) |- (midpoint_state);
    \draw[line, red, line width=2pt] (video.east) -- ++(1,0) node[midway, above, font=\footnotesize, text=red] {Look-ahead Info} |- (midpoint_state);

    \draw[arrow, line width=2pt] (midpoint_state) -- node[midway, align=center, fill=white, draw, thin, inner sep=4pt] {
        \textbf{Augmented State $s_k$}\\
        $\{\vec{x}_{net}, b_k, \vec{c}_{feat}, \textcolor{red}{\mathbf{\vec{v}_{next}}}\}$
    } ($(actor.west)!0.5!(critic.west) + (0, 1.5)$);

    \coordinate (agent_input) at ($(actor.west)!0.5!(critic.west) + (-0.5, 1.5)$);
    \draw[arrow] (agent_input) |- (actor.west);
    \draw[arrow] (agent_input) |- (critic.west);

    % VMAF-aware shield
    \node[block, fill=green!12, right=2.2cm of actor, minimum width=7.2em] (shield) {\textbf{VMAF-Aware}\\Shield $\mathcal{S}$};
    \draw[arrow, line width=1.5pt, blue!70!black] (actor.east) -- node[midway, above, font=\footnotesize, fill=white, inner sep=2pt] {$a^{\mathrm{raw}}_k$} (shield.west);
    \draw[arrow, line width=1.5pt, green!50!black] (shield.east) -- ++(0.6,0) |- node[pos=0.82, above, font=\footnotesize, fill=white, inner sep=2pt] {$a^{\mathrm{exec}}_k$} ($(buffer.west) - (0.9,0)$) -- (buffer.west);
    \draw[arrow, dashed, thick, orange!70!black] (buffer.south) -- ++(0,-0.5) -| node[pos=0.3, below, fill=white] {Reward $r_k$ (QoE)} ($(critic.south) - (0,0.5)$) -- (critic.south);
\end{tikzpicture}
