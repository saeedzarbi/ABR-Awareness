% TikZ-only system architecture: two-layer Policy--Shield co-design.
% Solid bold = inference path; dashed = training-only path.
\tikzset{
    archblock/.style={
        draw=black!60, rounded corners=2pt, align=center,
        minimum height=0.82cm, minimum width=2.55cm,
        inner sep=3.5pt, fill=white, font=\footnotesize
    },
    stateblock/.style={
        archblock, fill=orange!8, draw=orange!55!black,
        text width=2.55cm, minimum height=1.40cm
    },
    actorblock/.style={archblock, fill=red!9, draw=red!55!black},
    criticblock/.style={archblock, fill=blue!8, draw=blue!55!black},
    shieldblock/.style={
        archblock, fill=green!10, draw=green!45!black,
        minimum width=2.55cm, text width=2.45cm
    },
    groupbox/.style={
        draw=black!45, dashed, rounded corners=3pt,
        inner xsep=8pt, inner ysep=11pt
    },
    % Inference (deployed) path
    infer/.style={->, >=Latex, line width=1.25pt, solid, black!80},
    infergreen/.style={->, >=Latex, line width=1.35pt, solid, green!40!black},
    % Training-only path
    train/.style={->, >=Latex, line width=0.85pt, densely dashed, black!55},
    trainorange/.style={->, >=Latex, line width=0.85pt, densely dashed, orange!70!black},
    trainblue/.style={->, >=Latex, line width=0.85pt, densely dashed, blue!55!black},
}

\begin{tikzpicture}[x=1cm,y=1cm,font=\footnotesize]
    % Environment
    \node[archblock, fill=gray!8] (network) at (0,1.35)
        {Network channel\\Throughput \(C_k\)};
    \node[archblock, fill=gray!8] (buffer) at (0,0.05)
        {Client player\\Buffer \(B_k\)};
    \node[archblock, fill=red!5, draw=red!50!black] (video) at (0,-1.30)
        {Video source\\Chunk sizes / VMAF};

    % Observation aggregation and PPO
    \node[stateblock] (state) at (3.85,0.55)
        {\textbf{Augmented state \(s_k\)}\\[-1pt]
         throughput, buffer,\\prev.\ action, lookahead};
    \node[actorblock] (actor) at (7.20,1.05)
        {\textbf{Actor}\\Policy \(\pi_\theta(a\mid s)\)};
    \node[criticblock] (critic) at (7.20,-0.85)
        {\textbf{Critic}\\Value \(V_\phi(s)\)\\{\scriptsize (training only)}};
    \node[shieldblock] (shield) at (10.55,1.05)
        {\textbf{VMAF-aware shield}\\
         \(\mathcal{S}(s_k,a^{\mathrm{raw}}_k)\)};

    % Group boundaries
    \begin{pgfonlayer}{background}
        \node[groupbox, fill=blue!3, fit=(network) (buffer) (video)]
            (envgroup) {};
        \node[groupbox, fill=red!3, fit=(actor) (critic)]
            (agentgroup) {};
    \end{pgfonlayer}
    \node[font=\bfseries\footnotesize, text=blue!55!black, above=2pt of envgroup.north]
        {ABR streaming environment};
    \node[font=\bfseries\footnotesize, text=red!55!black, above=2pt of agentgroup.north]
        {Lagrangian PPO agent};

    % --- Observation → state (inference inputs) ---
    \draw[infer, gray!70] (network.east) -- ++(0.35,0)
        |- ([yshift=0.42cm]state.west);
    \draw[infer, gray!70] (buffer.east) -- (state.west);
    \draw[infer, red!60!black] (video.east) -- ++(0.35,0)
        |- node[pos=0.30, below, font=\scriptsize] {manifest / VMAF}
        ([yshift=-0.42cm]state.west);

    % --- Inference: state → actor → shield; state → shield ---
    \draw[infer] (state.east) --
        node[midway, above, font=\scriptsize, fill=white, inner sep=1pt] {\(s_k\)}
        (actor.west);
    \draw[infer] (actor.east) --
        node[midway, above, font=\scriptsize, fill=white, inner sep=1pt]
        {\(a^{\mathrm{raw}}_k\)} (shield.west);
    % Critical: shield also consumes \(s_k\) (buffer, throughput, VMAF ladder)
    \draw[infer, orange!65!black]
        (state.north) -- ++(0,0.95) -|
        node[pos=0.22, above, font=\scriptsize, fill=white, inner sep=1pt]
        {\(s_k\to\mathcal{S}\)}
        (shield.north);

    % --- Inference: executed action back to player ---
    \draw[infergreen]
        (shield.east) -- ++(0.55,0) -- ++(0,1.55) -|
        node[pos=0.55, above, font=\scriptsize, fill=white, inner sep=1pt]
        {\(a^{\mathrm{exec}}_k\)}
        (buffer.north);

    % --- Training only: state → critic; reward / Lyapunov cues → critic ---
    \draw[trainblue]
        (state.south) |-
        node[pos=0.75, below, font=\scriptsize, fill=white, inner sep=1pt]
        {\(s_k\) (train)}
        (critic.west);
    \draw[trainorange]
        (buffer.east) -- ++(0.55,0) |-
        node[pos=0.72, below, font=\scriptsize, fill=white, inner sep=1pt]
        {\(r_k,\;\phi,\;\psi^{\mathrm{Lya}}\)}
        (critic.south);

    % Legend
    \node[anchor=west, font=\scriptsize, align=left] at (3.2,-2.45) {
        \tikz\draw[infer] (0,0) -- (0.55,0);~\textbf{Inference} (deployed): \(s_k\!\to\!\pi_\theta\!\to\!a^{\mathrm{raw}}_k\),\;
        \(\mathcal{S}(s_k,a^{\mathrm{raw}}_k)\!\to\!a^{\mathrm{exec}}_k\)\\[1pt]
        \tikz\draw[train] (0,0) -- (0.55,0);~\textbf{Training only}: critic, reward \(r_k\), and Lagrangian cues \(\phi,\psi^{\mathrm{Lya}}\)
    };
\end{tikzpicture}
