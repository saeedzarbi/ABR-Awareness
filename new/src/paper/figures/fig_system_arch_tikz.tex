% Figure 1: simplified architecture — single left-to-right inference path,
% one training branch (Critic, dashed), one feedback arc. Scaled to \linewidth
% by \adjustbox in main.ltx, so absolute size here only fixes proportions.
\definecolor{archEnv}{HTML}{666666}
\definecolor{archState}{HTML}{E69F00}
\definecolor{archPolicy}{HTML}{0072B2}
\definecolor{archShield}{HTML}{009E73}

\tikzset{
    archbox/.style={
        draw, rounded corners=3pt, align=center, thick,
        inner sep=8pt, font=\normalsize, minimum height=1.25cm
    },
    envbox/.style={archbox, fill=archEnv!8, draw=archEnv, text width=3.1cm},
    statebox/.style={archbox, fill=archState!12, draw=archState!85!black, text width=2.3cm, minimum height=1.55cm},
    actorbox/.style={archbox, fill=archPolicy!12, draw=archPolicy, text width=2.3cm},
    shieldbox/.style={archbox, fill=archShield!14, draw=archShield!80!black, text width=2.7cm, minimum height=1.55cm},
    clientbox/.style={archbox, fill=archEnv!5, draw=archEnv, text width=2.0cm},
    criticbox/.style={archbox, thin, fill=archPolicy!4, draw=archPolicy!70, dash pattern=on 3pt off 2pt, text width=2.3cm},
    ppogroup/.style={draw=archPolicy!45, dash pattern=on 3pt off 2pt, rounded corners=5pt, inner xsep=10pt, inner ysep=10pt},
    infer/.style={->, line width=1.5pt, black!85},
    train/.style={->, line width=1.0pt, dash pattern=on 3pt off 2pt, archPolicy!70!black},
    feedback/.style={->, line width=1.05pt, archEnv},
    edgelabel/.style={font=\small, fill=white, inner sep=1.5pt},
}

\begin{tikzpicture}[font=\normalsize, >=Latex]

    % --- Main inference path (single row, left to right) ---
    \node[envbox] (env) at (0, 0)
        {\textbf{Environment}\\[3pt]{\footnotesize network throughput $C_k$, manifest/VMAF, buffer $B_k$}};

    \node[statebox] (state) at (3.7, 0)
        {\textbf{State} \(s_k\)\\[3pt]{\footnotesize throughput, buffer,}\\[0pt]{\footnotesize VMAF, lookahead}};

    \node[actorbox] (actor) at (6.9, 0)
        {\textbf{Base policy}\\[3pt]{\footnotesize heuristic or \(\pi_\theta\)}\\[0pt]{\footnotesize proposes \(a^{\mathrm{raw}}_k\)}};

    \node[shieldbox] (shield) at (10.3, 0)
        {\textbf{CPS}\\[3pt]{\footnotesize banking + conformal}\\[0pt]{\footnotesize \(\mathcal{S}(s_k,a^{\mathrm{raw}}_k)\)}};

    \node[clientbox] (client) at (13.5, 0)
        {\textbf{Client}\\[2pt]{\footnotesize player}};

    \draw[infer] (env.east) -- (state.west);
    \draw[infer] (state.east) -- (actor.west)
        node[edgelabel, midway, above=1pt] {\(s_k\)};
    \draw[infer] (actor.east) -- (shield.west)
        node[edgelabel, midway, above=1pt] {\(a^{\mathrm{raw}}_k\)};
    \draw[infer] (shield.east) -- (client.west)
        node[edgelabel, midway, above=1pt] {\(a^{\mathrm{exec}}_k\)};

    % --- Training-only branch (Critic), clearly set apart below the Actor ---
    \node[criticbox] (critic) at (6.9, -2.9)
        {\textbf{Critic} \(V_\phi(s)\)\\[2pt]{\footnotesize training only}\\[2pt]{\footnotesize reward \(r_k\), \(\lambda\)-update}};
    \draw[train] (state.south) -- ++(0,-0.45) -| (critic.north);

    \begin{pgfonlayer}{background}
        \node[ppogroup, fit=(actor)(critic)] (ppogrp) {};
    \end{pgfonlayer}

    \node[anchor=south west, font=\normalsize\bfseries, text=archPolicy]
        at (ppogrp.north east) {Optional RL host (PPO)};

    \draw[feedback]
        (client.north) -- ++(0,1.75) -| (env.north)
        node[edgelabel, font=\footnotesize, pos=0.5, above] {buffer update after chunk download};

    \node[font=\footnotesize, anchor=north] at (6.9, -4.55) {
        \tikz[baseline=-0.5ex]\draw[infer] (0,0) -- (0.6,0);~Inference \hspace{1.2em}
        \tikz[baseline=-0.5ex]\draw[train] (0,0) -- (0.6,0);~Training only \hspace{1.2em}
        \tikz[baseline=-0.5ex]\draw[feedback] (0,0) -- (0.6,0);~Download feedback
    };

\end{tikzpicture}
