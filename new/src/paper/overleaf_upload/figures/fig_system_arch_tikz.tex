% TikZ-only system architecture diagram.
\tikzset{
    archblock/.style={
        draw=black!60, rounded corners=2pt, align=center,
        minimum height=0.82cm, minimum width=2.65cm,
        inner sep=4pt, fill=white
    },
    stateblock/.style={
        archblock, fill=orange!8, draw=orange!55!black,
        text width=2.65cm, minimum height=1.35cm
    },
    actorblock/.style={archblock, fill=red!9, draw=red!55!black},
    criticblock/.style={archblock, fill=blue!8, draw=blue!55!black},
    shieldblock/.style={
        archblock, fill=green!10, draw=green!45!black,
        minimum width=2.45cm
    },
    groupbox/.style={
        draw=black!45, dashed, rounded corners=3pt,
        inner xsep=9pt, inner ysep=12pt
    },
    flow/.style={->, >=Latex, line width=0.85pt},
    feedback/.style={->, >=Latex, dashed, line width=0.8pt}
}

\begin{tikzpicture}[x=1cm,y=1cm,font=\small]
    % Environment
    \node[archblock, fill=gray!8] (network) at (0,1.25)
        {Network channel\\Throughput \(C_k\)};
    \node[archblock, fill=gray!8] (buffer) at (0,0)
        {Client player\\Buffer \(B_k\)};
    \node[archblock, fill=red!5, draw=red!50!black] (video) at (0,-1.25)
        {Video source\\Chunk sizes and VMAF};

    % Observation aggregation and PPO
    \node[stateblock] (state) at (3.85,0.55)
        {\textbf{Augmented state \(s_k\)}\\[-1pt]
         \(\{\)throughput, buffer,\\previous action, lookahead\(\}\)};
    \node[actorblock] (actor) at (7.15,0.9)
        {\textbf{Actor}\\Policy \(\pi_\theta(a\mid s)\)};
    \node[criticblock] (critic) at (7.15,-0.75)
        {\textbf{Critic}\\Value \(V_\phi(s)\)};
    \node[shieldblock] (shield) at (10.35,0.9)
        {\textbf{VMAF-aware shield}\\Projection \(\mathcal{S}\)};

    % Group boundaries and unobstructed labels
    \begin{pgfonlayer}{background}
        \node[groupbox, fill=blue!3, fit=(network) (buffer) (video)]
            (envgroup) {};
        \node[groupbox, fill=red!3, fit=(actor) (critic)]
            (agentgroup) {};
    \end{pgfonlayer}
    \node[font=\bfseries, text=blue!55!black, above=2pt of envgroup.north]
        {ABR streaming environment};
    \node[font=\bfseries, text=red!55!black, above=2pt of agentgroup.north]
        {Lagrangian PPO agent};

    % Observation paths
    \draw[flow, gray!75] (network.east) -- ++(0.38,0)
        |- ([yshift=0.43cm]state.west);
    \draw[flow, gray!75] (buffer.east) -- (state.west);
    \draw[flow, red!65!black] (video.east) -- ++(0.38,0)
        |- node[pos=0.28, below, font=\scriptsize] {manifest lookahead}
        ([yshift=-0.43cm]state.west);

    % Policy/value paths
    \draw[flow] (state.east) -- (actor.west);
    \draw[flow, dashed, black!65] (state.south) |- (critic.west);
    \draw[flow, blue!65!black] (actor.east) --
        node[midway, above, font=\scriptsize, fill=white, inner sep=1.5pt]
        {\(a^{\mathrm{raw}}_k\)} (shield.west);

    % Executed action returns to the player along a clear upper route
    \draw[flow, green!45!black, line width=1.05pt]
        (shield.north) -- ++(0,1.45) -|
        node[pos=0.48, above, font=\scriptsize, fill=white, inner sep=1.5pt]
        {\(a^{\mathrm{exec}}_k\)}
        (buffer.north);

    % Training reward to critic along a separate lower route
    \draw[feedback, orange!70!black]
        (buffer.east) -- ++(0.45,0) |-
        node[pos=0.78, below, font=\scriptsize, fill=white, inner sep=1.5pt]
        {reward \(r_k\)}
        (critic.south);
\end{tikzpicture}
